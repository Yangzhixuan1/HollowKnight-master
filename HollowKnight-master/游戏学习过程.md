## 二、角色移动

```c
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
public class playermovement : MonoBehaviour
{
    private Rigidbody2D rb; 
    //获取角色的 2D刚体组件。用于控制物理运动（如速度、重力、受力、碰撞响应等）。缓存后可以在 FixedUpdate 中直接修改 rb.velocity 等属性。
    private BoxCollider2D coll; 
    //获取角色的 2D盒状碰撞体组件。用于碰撞检测、触发事件，以及动态调整碰撞体的大小或偏移（例如角色蹲下时缩小碰撞体）。
    private Animator anim; 
    //获取角色的 动画控制器组件。用于播放、切换动画状态，设置动画参数（如 anim.SetBool("isRunning", true)）。
    private SpriteRenderer sprite;
    //获取角色的 精灵渲染器组件。用于控制角色图像的显示，比如翻转朝向（sprite.flipX = true）、改变颜色、调整渲染顺序等。
    private Vector2 initialColliderOffset;
    //用于存储 碰撞体的初始偏移量。当角色状态改变（如翻滚、下蹲）需要临时修改碰撞体位置或大小时，可以先用这个变量保存原始偏移，之后恢复原状，避免碰撞体错位。
    
    // 可跳跃地面的图层
    [SerializeField] private LayerMask jumpableGround;
    // 水平方向输入值
    private float dirX = 0f;
    // 可以改变移动速度
    [SerializeField] private float moveSpeed = 7f;
    // 可设置跳跃高度
    [SerializeField] private float jumpForce = 7f;
    // 动画状态枚举：待机、奔跑、跳跃、下落
    private enum MovementState 
    {
        idle,
        running,
        jumping, 
        falling 
    }

    // 跳跃音效
    [SerializeField] private AudioSource jumpSoundEffect;
    void Start()
    {
        // 获取 Rigidbody2D 组件
        rb = GetComponent<Rigidbody2D>();
        // 获取 BoxCollider2D 组件
        coll = GetComponent<BoxCollider2D>();
        // 获取 SpriteRenderer 组件
        sprite = GetComponent<SpriteRenderer>();
        // 获取 Animator 组件
        anim = GetComponent<Animator>();
        
        initialColliderOffset = coll.offset;
    }

    // 每帧调用一次
    private void Update()
    {
        // 读取水平输入，-1、0、1
        dirX = Input.GetAxisRaw("Horizontal");
        // 保持 y 轴速度不变，设置水平移动速度
        rb.velocity = new Vector2(dirX * moveSpeed, rb.velocity.y);

        // 按下跳跃键且在地面上时
        if (Input.GetButtonDown("Jump") && isGrounded())
        {
            // 保持 x 轴速度不变，设置跳跃速度
            rb.velocity = new Vector2(rb.velocity.x, jumpForce);
            // 播放跳跃音效
            jumpSoundEffect.Play();
        }
        // 更新动画状态
        UpdateAnimationStatate();
    }
    // 根据移动与垂直速度更新动画状态
    private void UpdateAnimationStatate()
    {
        // 声明当前动画状态
        MovementState state;

        // 如果向右移动
        if (dirX > 0f)
        {
            // 状态设为奔跑
            state = MovementState.running;
            // 精灵不翻转
            sprite.flipX = false;
            //碰撞器不反转
            coll.offset = new Vector2(initialColliderOffset.x, initialColliderOffset.y);
        }
        // 如果向左移动
        else if (dirX < 0f)
        {
            // 状态设为奔跑
            state = MovementState.running;
            // 精灵水平翻转
            sprite.flipX = true;
            //碰撞器反转
            coll.offset = new Vector2(-initialColliderOffset.x, initialColliderOffset.y);
        }
        // 如果没有水平移动
        else
        {
            // 状态设为待机
            state = MovementState.idle;
        }

        // 如果 y 轴速度大于 0.1，说明正在上升
        if (rb.velocity.y > .1f)
        {
            // 状态设为跳跃
            state = MovementState.jumping;
        }
        // 如果 y 轴速度小于 -0.1，说明正在下落
        else if (rb.velocity.y < -.1f)
        {
            // 状态设为下落
            state = MovementState.falling;
        }

        // 将枚举状态转换为整数，设置到 Animator 的 state 参数
        anim.SetInteger("state", (int)state);
    }

    // 检测玩家是否在地面上
    private bool isGrounded()
    {
        // 使用 BoxCast 从碰撞体中心向下检测，碰到可跳跃地面层则返回 true
        return Physics2D.BoxCast(coll.bounds.center, coll.bounds.size, 0f, Vector2.down, .1f, jumpableGround);
    }
}
```

## 三、音频管理
### 1. audio source
#### 组件内容
1. AudioClip 放置自己的音频片段
2. Ouput 使用什么混响器
3. Mute 是否静音
4. Play On Awake 开局自动播放
5. Loop 循环播放
6. Priority 声音优先级越小越高级
7. Volume 控制基础音量
8. Pitch 音调同时会改变播放速度
9. Stero Pan 左右声道
10. Spatial Blend 选择2D还是3D音频
11. Revern Zone Mix 声音会受到环境影响程度

#### 函数部分
audioSource.Play(); //直接播放
audioSource.PlayOneShot(MusicClip); //播放一次
audioSource.Stop(); //停止音频播放
audioSource.Pause(); //暂停
audioSource.UnPause(); //解除暂停
audioSource.PlayDelayed(1f); //延迟指定秒后播放

``` c#
public class ShootSystem : MonoBehaviour
{
    public float rayDistance = 100f;       //射线检测的最大距离

    private GameObject currentTarget;

    private void Update()
    {
        // 从摄像机画面正中心发射射线
        Ray ray = Camera.main.ViewportPointToRay(
            new Vector3(0.5f, 0.5f, 0f)
        );

        RaycastHit hit;

        // 检测射线是否碰到物体
        if (Physics.Raycast(ray, out hit, rayDistance))
        {
            currentTarget = hit.collider.gameObject;

            // 按下鼠标左键时，将瞄准的物体变成绿色
            if (Input.GetMouseButtonDown(0))
            {
                EnemySystem.enemySystem.DefeatEnemy(currentTarget);
            }
        }
        else
        {
            // 射线没有碰到物体
            currentTarget = null;
        }

        // 在 Scene 视图中绘制射线，方便调试
        Debug.DrawRay(ray.origin, ray.direction * rayDistance, Color.red);
    }
}
```
